% Custom commands
\newcommand{\resumeItem}[1]{
  \item\small{
    {#1 \vspace{-2pt}}
  }
}

% Optional tech stack: \resumeSubheading[Tech]{Role}{Date}{Org}{Location}
% Without tech:      \resumeSubheading{Role}{Date}{Org}{Location}
\NewDocumentCommand{\resumeSubheading}{o m m m m}{%
  \vspace{0pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \IfNoValueTF{#1}{%
        \small#2%
      }{%
        \small#2 \small{$|$ #1}%
      } & \small#3 \\
      \textit{\small#4} & \textit{\small #5} \\
    \end{tabular*}\vspace{-9pt}%
}
